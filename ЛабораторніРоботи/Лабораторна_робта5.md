**Короткі теоретичні відомості**

**Прототип сервісу** — це початкова реалізація одного мікросервісу, яка дозволяє перевірити структуру API та взаємодію з базою даних.

**REST API**

REST — це стиль побудови веб-сервісів, у якому використовуються стандартні HTTP-методи:

- GET — отримання даних
- POST — створення
- PUT — оновлення
- DELETE — видалення

Дані зазвичай передаються у форматі JSON.

**CRUD-операції**

CRUD складається з чотирьох базових дій:

- Create — створення
- Read — читання
- Update — оновлення
- Delete — видалення

**PostgreSQL**

Система керування базами даних, яка використовується для зберігання структурованої інформації.

**Docker**

Інструмент для запуску застосунків у контейнерах. Для мікросервісу створюються:

- Dockerfile — опис контейнера
- docker-compose — запуск сервісу + бази даних

**main.py**

```python
from fastapi import FastAPI, Depends, HTTPException
from sqlalchemy import create_engine, Column, Integer, String
from sqlalchemy.orm import sessionmaker, declarative_base, Session
from sqlalchemy.exc import OperationalError
from pydantic import BaseModel
import time


DATABASE_URL = "postgresql://postgres:postgres@db:5432/construction_db"

engine = create_engine(DATABASE_URL)
SessionLocal = sessionmaker(bind=engine)
Base = declarative_base()


class ConstructionObject(Base):
    __tablename__ = "construction_objects"

    id = Column(Integer, primary_key=True, index=True)
    name = Column(String, nullable=False)
    location = Column(String, nullable=False)
    status = Column(String, nullable=False)

# ======================
# Pydantic schemas
# ======================
class ConstructionBase(BaseModel):
    name: str
    location: str
    status: str

class ConstructionCreate(ConstructionBase):
    pass

class Construction(ConstructionBase):
    id: int

    class Config:
        from_attributes = True


app = FastAPI(title="Construction Service")


@app.on_event("startup")
def startup_event():
    retries = 5
    while retries > 0:
        try:
            Base.metadata.create_all(bind=engine)
            print("Database connected successfully")
            break
        except OperationalError:
            print("Database not ready, retrying...")
            retries -= 1
            time.sleep(3)

def get_db():
    db = SessionLocal()
    try:
        yield db
    finally:
        db.close()

# ======================
# CRUD endpoints
# ======================
@app.post("/objects", response_model=Construction)
def create_object(obj: ConstructionCreate, db: Session = Depends(get_db)):
    new_obj = ConstructionObject(**obj.dict())
    db.add(new_obj)
    db.commit()
    db.refresh(new_obj)
    return new_obj

@app.get("/objects", response_model=list[Construction])
def get_objects(db: Session = Depends(get_db)):
    return db.query(ConstructionObject).all()

@app.get("/objects/{object_id}", response_model=Construction)
def get_object(object_id: int, db: Session = Depends(get_db)):
    obj = db.query(ConstructionObject).filter_by(id=object_id).first()
    if not obj:
        raise HTTPException(status_code=404, detail="Object not found")
    return obj

@app.put("/objects/{object_id}", response_model=Construction)
def update_object(object_id: int, updated: ConstructionCreate, db: Session = Depends(get_db)):
    obj = db.query(ConstructionObject).filter_by(id=object_id).first()
    if not obj:
        raise HTTPException(status_code=404, detail="Object not found")

    obj.name = updated.name
    obj.location = updated.location
    obj.status = updated.status

    db.commit()
    db.refresh(obj)
    return obj

@app.delete("/objects/{object_id}")
def delete_object(object_id: int, db: Session = Depends(get_db)):
    obj = db.query(ConstructionObject).filter_by(id=object_id).first()
    if not obj:
        raise HTTPException(status_code=404, detail="Object not found")

    db.delete(obj)
    db.commit()
    return {"message": "Object deleted"}

```
